    <script>
        const APP_KEY = 'BD_TASK_INCOME_SECURE_DB_2026';
        
        function getDB() { 
            const raw = localStorage.getItem(APP_KEY);
            let db = raw ? JSON.parse(raw) : null;
            if(!db) {
                db = { 
                    settings: { bkashNumber: '01711223344', nagadNumber: '01811223344', telegramLink: 'https://t.me/', whatsappLink: 'https://wa.me/' }, 
                    users: [], tasks: [], activations: [], withdrawals: [] 
                };
                localStorage.setItem(APP_KEY, JSON.stringify(db));
            }
            return db;
        }

        function saveDB(db) { localStorage.setItem(APP_KEY, JSON.stringify(db)); }

        let currentUser = null;

        window.addEventListener('DOMContentLoaded', () => {
            const urlParams = new URLSearchParams(window.location.search);
            const refParam = urlParams.get('ref');
            if (refParam) {
                switchAuthTab('register');
                document.getElementById('regRefCode').value = refParam;
            } else {
                const savedUserId = sessionStorage.getItem('active_user_id');
                if (savedUserId) {
                    const db = getDB();
                    currentUser = db.users.find(u => u.id === savedUserId);
                    if (currentUser) showSection('dashboardPage');
                }
            }
            updateNav();
        });

        function updateNav() {
            const nav = document.getElementById('navRight');
            if (currentUser) {
                nav.innerHTML = `<span class="font-bold text-xs">${currentUser.name.split(' ')[0]}</span><button onclick="logout()" class="text-rose-600 ml-2 font-bold"><i class="fa-solid fa-right-from-bracket"></i></button>`;
            } else { nav.innerHTML = ''; }
        }

        function logout() { sessionStorage.removeItem('active_user_id'); location.reload(); }
        
        function switchAuthTab(tab) {
            document.getElementById('loginFormContainer').classList.toggle('hidden', tab !== 'login');
            document.getElementById('registerFormContainer').classList.toggle('hidden', tab === 'login');
        }

        function handleUserLogin(e) {
            e.preventDefault();
            const db = getDB();
            const user = db.users.find(u => u.phone === document.getElementById('loginPhone').value && u.passwordHash === document.getElementById('loginPassword').value);
            if (user) { currentUser = user; sessionStorage.setItem('active_user_id', user.id); showSection('dashboardPage'); updateNav(); }
            else alert('ভুল মোবাইল নম্বর বা পাসওয়ার্ড!');
        }

        function handleUserRegister(e) {
            e.preventDefault();
            const db = getDB();
            const phone = document.getElementById('regPhone').value;
            if (db.users.some(u => u.phone === phone)) { alert('এই নম্বরে আগেই অ্যাকাউন্ট খোলা হয়েছে!'); return; }
            const newUser = { id: 'usr_' + Date.now(), name: document.getElementById('regName').value, phone, passwordHash: document.getElementById('regPassword').value, status: 'Pending Activation', balance: 0, pendingBalance: 0, refCode: 'BD' + Math.floor(100000 + Math.random() * 900000) };
            db.users.push(newUser); saveDB(db); currentUser = newUser; sessionStorage.setItem('active_user_id', newUser.id);
            showSection('activationPage'); updateNav();
        }

        function showSection(id) {
            ['authPage', 'dashboardPage', 'activationPage', 'taskCenterPage', 'referEarnPage', 'walletPage'].forEach(s => document.getElementById(s)?.classList.add('hidden'));
            document.getElementById(id)?.classList.remove('hidden');
            
            const db = getDB();
            
            if (id === 'dashboardPage') {
                document.getElementById('userBalanceDisplay').innerText = `৳${currentUser.balance.toFixed(2)}`;
                document.getElementById('userPendingDisplay').innerText = `৳${currentUser.pendingBalance.toFixed(2)}`;
                document.getElementById('activationBanner').classList.toggle('hidden', currentUser.status === 'Active');
                
                // Update support links
                document.getElementById('usrTelegramLink').href = db.settings.telegramLink || '#';
                document.getElementById('usrWhatsappLink').href = db.settings.whatsappLink || '#';
            }
            
            if (id === 'activationPage') {
                document.getElementById('usrBkashNum').innerText = db.settings.bkashNumber || '017XXXXXXXX';
                document.getElementById('usrNagadNum').innerText = db.settings.nagadNumber || '018XXXXXXXX';
            }
            
            if (id === 'referEarnPage') document.getElementById('userRefLinkDisplay').value = `${window.location.origin}${window.location.pathname}?ref=${currentUser.refCode}`;
            if (id === 'taskCenterPage') renderTasks();
        }

        function copyElementText(elementId) {
            const txt = document.getElementById(elementId).innerText;
            navigator.clipboard.writeText(txt);
            alert('নম্বরটি কপি হয়েছে: ' + txt);
        }

        function renderTasks() {
            const db = getDB();
            document.getElementById('taskList').innerHTML = db.tasks.map(t => `
                <div class="p-3 border rounded-xl flex justify-between items-center bg-white">
                    <div><b>${t.title}</b><br><span class="text-emerald-600 font-bold">৳${t.reward}</span></div>
                    <a href="${t.link}" target="_blank" class="bg-sky-900 text-white px-3 py-1 rounded-lg font-bold">ভিজিট করুন</a>
                </div>
            `).join('') || '<p class="text-slate-400 text-center py-4">কোনো টাস্ক নেই</p>';
        }

        function copyRefLink() {
            navigator.clipboard.writeText(document.getElementById('userRefLinkDisplay').value);
            alert('রেফারেল লিংক কপি হয়েছে!');
        }

        function submitActivationRequest(e) {
            e.preventDefault();
            const db = getDB();
            db.activations.push({ id: 'act_' + Date.now(), userId: currentUser.id, userName: currentUser.name, method: document.getElementById('actMethod').value, senderPhone: document.getElementById('actSenderPhone').value, trxId: document.getElementById('actTrxId').value, status: 'Pending' });
            saveDB(db); alert('অ্যাক্টিভেশন রিকোয়েস্ট জমা হয়েছে!'); showSection('dashboardPage');
        }

        function handleWithdrawRequest(e) {
            e.preventDefault();
            const amt = parseFloat(document.getElementById('wDrawAmount').value);
            if (amt > currentUser.balance) { alert('অপর্যাপ্ত ব্যালেন্স!'); return; }
            const db = getDB();
            currentUser.balance -= amt;
            db.users.find(u => u.id === currentUser.id).balance = currentUser.balance;
            db.withdrawals.push({ id: 'wdr_' + Date.now(), userName: currentUser.name, phone: document.getElementById('wDrawPhone').value, amount: amt, status: 'Pending' });
            saveDB(db); alert('উইথড্র রিকোয়েস্ট সফল হয়েছে!'); showSection('dashboardPage');
        }
    </script>
